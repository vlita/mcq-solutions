\item[c.] If $\mathbf{U}$ is unitary and $\mathbf{B} = \mathbf{U}^{\dagger}\mathbf{AU}$, then $\mathbf{A} = \mathbf{UBU}^{\dagger}$.

\item[e.] If $\mathbf{A}$ is Hermitian then $\mathbf{A}^{-1}$, if it exists, is also Hermitian.

\begin{pmatrix}
A_{22} & -A_{12} \\
-A_{21} & A_{11}
\end{pmatrix}.
\]
\end{enumerate}
